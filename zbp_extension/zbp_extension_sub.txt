FORM download_template .

  DATA lv_file TYPE string.
  DATA gt_excel TYPE STANDARD TABLE OF ty_customer_data.

  lv_file = p_file.

  IF p_file IS INITIAL.
    MESSAGE 'Enter File Path' TYPE sy-abcde+18(1) DISPLAY LIKE sy-abcde+4(1).
    LEAVE LIST-PROCESSING.
  ENDIF.

  gt_excel = VALUE #( (
   kunnr = 'Customer'
   vkorg = 'Sales_Org'
   vtweg = 'Dist_Channel'
   spart = 'Division'
   kalks =  'Pricing Procedure'
   kdgrp = 'Customer Group'
   vwerk = 'Delivery Plant'
   vsbed = 'Shiping Condition'
   inco1 = 'Incoterm'
   inco2 = 'Incoterm2'
   zterm = 'Payment Term'
   ktgrd = 'Account assignment group'
   kvgr1 = 'Customer Grp1'
   kvgr2 = 'Customer Grp2'
   kvgr3 = 'Customer Grp3'
   ) ).

*  Download Excel Template
  CALL FUNCTION 'GUI_DOWNLOAD'
    EXPORTING
      filename                = lv_file
      filetype                = 'ASC'
      write_field_separator   = abap_true
    TABLES
      data_tab                = gt_excel
    EXCEPTIONS
      file_write_error        = 1
      no_batch                = 2
      gui_refuse_filetransfer = 3
      invalid_type            = 4
      no_authority            = 5
      unknown_error           = 6
      header_not_allowed      = 7
      separator_not_allowed   = 8
      filesize_not_allowed    = 9
      header_too_long         = 10
      dp_error_create         = 11
      dp_error_send           = 12
      dp_error_write          = 13
      unknown_dp_error        = 14
      access_denied           = 15
      dp_out_of_memory        = 16
      disk_full               = 17
      dp_timeout              = 18
      file_not_found          = 19
      dataprovider_exception  = 20
      control_flush_error     = 21
      OTHERS                  = 22.

  IF sy-subrc <> 0.
    MESSAGE TEXT-e02 TYPE sy-abcde+18(1) DISPLAY LIKE sy-abcde+4(1).
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
FORM upload_file_to_itab .

  DATA : filename TYPE rlgrap-filename,
         it_type  TYPE truxs_t_text_data.

  filename = p_file.


  IF p_mode <> 'A' AND p_mode <> 'E' AND p_mode <> 'N' AND p_mode <> 'P'.
    MESSAGE 'Enter Valid Processing Mode (A, E, N, P)' TYPE sy-abcde+18(1) DISPLAY LIKE sy-abcde+4(1).
    LEAVE LIST-PROCESSING.
  ENDIF.

  IF p_file IS INITIAL.
    MESSAGE 'Enter File Path' TYPE sy-abcde+18(1) DISPLAY LIKE sy-abcde+4(1).
    LEAVE LIST-PROCESSING.
  ENDIF.

* Uploading the data from the file into the internal table
  CALL FUNCTION 'TEXT_CONVERT_XLS_TO_SAP'
    EXPORTING
      i_line_header        = abap_true
      i_tab_raw_data       = it_type
      i_filename           = filename
    TABLES
      i_tab_converted_data = gt_data
    EXCEPTIONS
      conversion_failed    = 1
      OTHERS               = 2.

  IF sy-subrc NE 0.
    MESSAGE TEXT-e02 TYPE sy-abcde+18(1) DISPLAY LIKE sy-abcde+4(1).
    LEAVE LIST-PROCESSING.
  ENDIF.

  IF lines( gt_data ) = 0.
    MESSAGE TEXT-e03 TYPE sy-abcde+18(1) DISPLAY LIKE sy-abcde+4(1).
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form fill_alv
*&---------------------------------------------------------------------*
*& fill_alv
*&---------------------------------------------------------------------*
*&      --> LS_EXCEL
*&---------------------------------------------------------------------*
FORM fill_alv USING im_excel TYPE ty_customer_data.

  DATA lv_message(100) TYPE c.

  LOOP AT gt_messtab INTO DATA(ls_messtab).

    CALL FUNCTION 'FORMAT_MESSAGE'
      EXPORTING
        id        = ls_messtab-msgid
        lang      = sy-langu
        no        = ls_messtab-msgnr
        v1        = ls_messtab-msgv1
        v2        = ls_messtab-msgv2
        v3        = ls_messtab-msgv3
        v4        = ls_messtab-msgv4
      IMPORTING
        msg       = lv_message
      EXCEPTIONS
        not_found = 1
        OTHERS    = 2.

    IF sy-subrc = 0.

      APPEND VALUE #( serial_no    = 1
         Customer = im_excel-kunnr
      message      = lv_message
      ) TO gt_log.

      CLEAR lv_message.

    ENDIF.

  ENDLOOP.

  CLEAR: gt_bdcdata,
  gt_messtab.

ENDFORM.
FORM bdc_dynpro USING program TYPE bdcdata-program
      dynpro  TYPE bdcdata-dynpro.

  gt_bdcdata = VALUE #( BASE gt_bdcdata ( program  = program
  dynpro   = dynpro
  dynbegin = abap_true ) ).

ENDFORM.
FORM bdc_field USING fnam TYPE any
      fval TYPE any.

  gt_bdcdata = VALUE #( BASE gt_bdcdata ( fnam = fnam
  fval = fval ) ).

ENDFORM.
FORM bdc_transaction USING tcode TYPE tcode.

  DATA(ls_options) = VALUE ctu_params( dismode = p_mode       "Screen Mode
        updmode = 'A'          "Asynchronous
        defsize = abap_true
        nobinpt = abap_true ).

  CALL TRANSACTION tcode USING gt_bdcdata
        OPTIONS FROM ls_options
        MESSAGES INTO gt_messtab.
ENDFORM.

      t_outtab           = gt_log
